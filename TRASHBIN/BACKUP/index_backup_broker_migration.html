        initResizer() {
          const resizer = document.getElementById("pm-resizer");
          if (!resizer) return;

          let isDragging = false;
          let startY = 0;
          let startHeight = 0;

          const onMouseDown = (e) => {
            isDragging = true;
            startY = e.clientY || (e.touches && e.touches[0].clientY);
            startHeight = this.height;
            document.body.style.cursor = "ns-resize";
            document.body.style.userSelect = "none";
            document.getElementById("position-manager-panel")?.classList.add("pm-resizing");
            e.preventDefault();
          };

          const onMouseMove = (e) => {
            if (!isDragging) return;
            const currentY = e.clientY || (e.touches && e.touches[0].clientY);
            const deltaY = startY - currentY;
            let targetHeight = startHeight + deltaY;
            targetHeight = Math.max(this.minHeight, targetHeight);
            const maxAllowed = Math.max(this.minHeight, window.innerHeight - 140);
            targetHeight = Math.min(targetHeight, maxAllowed);

            this.isCollapsed = false;
            document.getElementById("position-manager-panel")?.classList.remove("pm-collapsed");
            const collapseIcon = document.getElementById("pm-collapse-icon");
            if (collapseIcon) collapseIcon.innerText = "▼";

            this.setHeight(targetHeight);
          };

          const onMouseUp = () => {
            if (isDragging) {
              isDragging = false;
              document.body.style.cursor = "";
              document.body.style.userSelect = "";
              document.getElementById("position-manager-panel")?.classList.remove("pm-resizing");
              localStorage.setItem("pm_panel_height", this.height);
              localStorage.setItem("pm_panel_collapsed", "false");
              window.dispatchEvent(new Event("resize"));
            }
          };

          resizer.addEventListener("mousedown", onMouseDown);
          resizer.addEventListener("touchstart", onMouseDown, { passive: false });
          document.addEventListener("mousemove", onMouseMove);
          document.addEventListener("touchmove", onMouseMove, { passive: false });
          document.addEventListener("mouseup", onMouseUp);
          document.addEventListener("touchend", onMouseUp);

          resizer.addEventListener("dblclick", () => {
            this.toggleCollapse();
          });
        },

        async fetchPositions() {
          try {
            const res = await fetch(`${this.backendUrl}/trade/positions`, {
              signal: AbortSignal.timeout ? AbortSignal.timeout(1500) : undefined
            });
            if (res.ok) {
              const json = await res.json();
              const items = Array.isArray(json) ? json : (json.positions || []);
              if (Array.isArray(items)) {
                this.data.positions = items.map(p => this.normalizePosition(p));
                this.updateAccountMetrics();
                this.renderPositions();
                this.renderAccountBar();
              }
            }
          } catch(e) {}
        },

        async fetchOrders() {
          try {
            const res = await fetch(`${this.backendUrl}/trade/orders`, {
              signal: AbortSignal.timeout ? AbortSignal.timeout(1500) : undefined
            });
            if (res.ok) {
              const json = await res.json();
              const items = Array.isArray(json) ? json : (json.orders || []);
              if (Array.isArray(items)) {
                this.data.orders = items.map(o => this.normalizeOrder(o));
                this.renderOrders();
              }
            }

        async fetchAccount() {
          try {
            const res = await fetch(`${this.backendUrl}/trade/account`, {
              signal: AbortSignal.timeout ? AbortSignal.timeout(1500) : undefined
            });
            if (res.ok) {
              const acc = await res.json();
              if (acc && typeof acc.balance === "number") {
                this.data.account = {
                  balance: acc.balance,
                  equity: acc.equity !== undefined ? acc.equity : acc.balance,
                  margin: acc.margin || 0,
                  free_margin: acc.free_margin !== undefined ? acc.free_margin : (acc.balance - (acc.margin || 0)),

                  profit: acc.profit || 0,
                  currency: acc.currency || "USD"
                };
                this.renderAccountBar();
              }
            }
          } catch(e) {}
        },

        normalizePosition(p) {
          return {
            ticket: p.ticket || p.id || Math.floor(Math.random() * 1000000),
            symbol: p.symbol || "XAUUSD.",
            type: (p.type === 0 || String(p.type).toUpperCase() === "BUY") ? "BUY" : "SELL",
            volume: parseFloat(p.volume || p.lots || 0.01),
            open_price: parseFloat(p.open_price || p.price_open || 0),
            current_price: parseFloat(p.current_price || p.price_current || p.open_price || 0),
            sl: parseFloat(p.sl || 0),
            tp: parseFloat(p.tp || 0),
            profit: parseFloat(p.profit || 0),
            pips: parseFloat(p.pips || 0),
            open_time: p.open_time || p.time || new Date().toISOString().replace('T', ' ').substring(0, 19)
          };
        },

        normalizeOrder(o) {
          let typeStr = String(o.type || "").toUpperCase();
          if (o.type === 2) typeStr = "BUY LIMIT";
          else if (o.type === 3) typeStr = "SELL LIMIT";
          else if (o.type === 4) typeStr = "BUY STOP";
          else if (o.type === 5) typeStr = "SELL STOP";
          if (!typeStr) typeStr = "BUY LIMIT";

          return {
            ticket: o.ticket || o.id || Math.floor(Math.random() * 1000000),
            symbol: o.symbol || "XAUUSD.",
            type: typeStr,
            volume: parseFloat(o.volume || o.lots || 0.01),
            price: parseFloat(o.price || o.price_open || 0),
            sl: parseFloat(o.sl || 0),
            tp: parseFloat(o.tp || 0),
            time: o.time || new Date().toISOString().replace('T', ' ').substring(0, 19)
          };
        },

        normalizeHistory(h) {
          return {
            ticket: h.ticket || h.order || h.deal || Math.floor(Math.random() * 1000000),
            time: h.time || h.close_time || new Date().toISOString().replace('T', ' ').substring(0, 19),
            symbol: h.symbol || "XAUUSD.",
            type: (h.type === 0 || String(h.type).toUpperCase() === "BUY") ? "BUY" : "SELL",
            volume: parseFloat(h.volume || h.lots || 0.01),
            open_price: parseFloat(h.open_price || h.price_open || 0),
            close_price: parseFloat(h.close_price || h.price_close || h.price || 0),

          };
        },

        updateLiveQuote(symbol, bid, ask) {
          if (!symbol || !bid || !ask) return;
          const cleanSym = symbol.toUpperCase().replace(/\./g, '');
          let changed = false;

          this.data.positions.forEach(pos => {
            const posSym = pos.symbol.toUpperCase().replace(/\./g, '');
            if (posSym === cleanSym) {
              const contract = this.contractSizes[pos.symbol] || (cleanSym.includes('XAU') ? 100 : 100000);
              const curPrice = pos.type === "BUY" ? bid : ask;
              pos.current_price = curPrice;
              const point = (cleanSym.includes('JPY') || cleanSym.includes('XAU')) ? 0.01 : 0.00001;
              const pipSize = point * 10;

              if (pos.type === "BUY") {
                pos.profit = (curPrice - pos.open_price) * pos.volume * contract;
                pos.pips = (curPrice - pos.open_price) / pipSize;
              } else {
                pos.profit = (pos.open_price - curPrice) * pos.volume * contract;
                pos.pips = (pos.open_price - curPrice) / pipSize;
              }
              changed = true;

              // Fast cell DOM updates without re-rendering entire table
              const priceCell = document.getElementById(`pm-price-${pos.ticket}`);
              const pnlCell = document.getElementById(`pm-pnl-${pos.ticket}`);
              if (priceCell) priceCell.innerText = this.formatPrice(curPrice, pos.symbol);
              if (pnlCell) {
                const pnlSign = pos.profit > 0 ? "+" : "";
                const pipsSign = pos.pips > 0 ? "+" : "";
                pnlCell.innerHTML = `${pnlSign}$${pos.profit.toFixed(2)} (${pipsSign}${pos.pips.toFixed(1)} pips)`;
                pnlCell.className = `pm-num-col ${pos.profit >= 0 ? 'pm-pnl-positive' : 'pm-pnl-negative'}`;
              }
            }
          });

          if (changed) {
            this.updateAccountMetrics();
            this.renderAccountBar();
          }
        },

        updateAccountMetrics() {
          let totalFloatingPnl = 0;
          let totalMargin = 0;

          this.data.positions.forEach(p => {
            totalFloatingPnl += (p.profit || 0);
            totalMargin += (p.volume || 0.1) * 200;
          });

          this.data.account.profit = totalFloatingPnl;
          this.data.account.equity = this.data.account.balance + totalFloatingPnl;
          this.data.account.margin = totalMargin;
          this.data.account.free_margin = this.data.account.equity - totalMargin;
          this.data.account.margin_level = totalMargin > 0 
            ? (this.data.account.equity / totalMargin) * 100 
            : null;
        },

No Trade History

                  
Closed deals and realized profits will appear here

                
              
            `;
            return;
          }

          let html = "";
          this.data.history.forEach(hist => {
            const isBuy = hist.type === "BUY";
            const badgeClass = isBuy ? "pm-badge-buy" : "pm-badge-sell";
            const pnlColorClass = hist.profit >= 0 ? "pm-pnl-positive" : "pm-pnl-negative";

                ${hist.volume.toFixed(2)}

        renderAccountBar() {
          const acc = this.data.account;
          const balEl = document.getElementById("pm-acc-balance");
          const eqEl = document.getElementById("pm-acc-equity");
          const marEl = document.getElementById("pm-acc-margin");
          const fmarEl = document.getElementById("pm-acc-freemargin");
          const mlvlEl = document.getElementById("pm-acc-marginlevel");
          const pnlEl = document.getElementById("pm-acc-pnl");

          if (balEl) balEl.innerText = `$${acc.balance.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
          if (eqEl) eqEl.innerText = `$${acc.equity.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
          if (marEl) marEl.innerText = `$${acc.margin.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
          if (fmarEl) fmarEl.innerText = `$${acc.free_margin.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;

          if (mlvlEl) {
            mlvlEl.innerText = acc.margin_level !== null && !isNaN(acc.margin_level) 
              ? `${acc.margin_level.toFixed(1)}%` 
              : "--";
          }

          if (pnlEl) {
            const sign = acc.profit > 0 ? "+" : "";
            const pct = acc.balance > 0 ? (acc.profit / acc.balance) * 100 : 0;
            pnlEl.innerText = `${sign}$${acc.profit.toFixed(2)} (${sign}${pct.toFixed(2)}%)`;
            pnlEl.className = "pm-metric-val " + (acc.profit > 0 ? "pm-pnl-positive" : acc.profit < 0 ? "pm-pnl-negative" : "pm-pnl-zero");
          }
        },

        formatPrice(price, symbol) {
          if (isNaN(price)) return "0.00";
          const s = (symbol || "").toUpperCase();
          if (s.includes("JPY")) return price.toFixed(3);
          if (s.includes("XAU") || s.includes("BTC") || s.includes("GOLD") || s.includes("INDEX")) return price.toFixed(2);
          if (s.includes("EUR") || s.includes("GBP") || s.includes("AUD") || s.includes("NZD") || s.includes("USD")) return price.toFixed(5);
          return price.toFixed(2);
        },

        async closePosition(ticket) {
          try {
            const res = await fetch(`${this.backendUrl}/trade/close`, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({ ticket: Number(ticket) }),
              signal: AbortSignal.timeout ? AbortSignal.timeout(3000) : undefined
            });
            if (res.ok) {
              this.showToast(`Position #${ticket} closed successfully`, 'success');
              await this.refresh();
              return true;
            }
          } catch(e) {}

          // Resilient fallback: update in-memory state and move to trade history
          const idx = this.data.positions.findIndex(p => p.ticket == ticket);
          if (idx !== -1) {
            const pos = this.data.positions.splice(idx, 1)[0];
            const nowTime = new Date().toISOString().replace('T', ' ').substring(0, 19);
            this.data.history.unshift({
              ticket: pos.ticket,
              time: nowTime,
              symbol: pos.symbol,
              type: pos.type,
              volume: pos.volume,
              open_price: pos.open_price,
              close_price: pos.current_price,
              profit: pos.profit
            });

            this.data.account.balance += pos.profit;
            this.updateAccountMetrics();
            this.renderPositions();
            this.renderHistory();
            this.renderAccountBar();

            const pnlSign = pos.profit >= 0 ? "+" : "";
            this.showToast(`Position #${ticket} closed (${pnlSign}$${pos.profit.toFixed(2)})`, 'success');
            return true;
          }
          return false;
        },

        async closeAllPositions() {
          if (this.data.positions.length === 0) {
            this.showToast("No active open positions to close", "info");
            return;
          }

          if (!confirm(`Are you sure you want to close all ${this.data.positions.length} open position(s)?`)) {
            return;
          }

          try {
            const res = await fetch(`${this.backendUrl}/trade/close_all`, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              signal: AbortSignal.timeout ? AbortSignal.timeout(4000) : undefined
            });
            if (res.ok) {
              this.showToast("All open positions closed successfully", "success");
              await this.refresh();
              return;
            }
          } catch(e) {}

          // Fallback: batch close in memory
          let totalClosed = 0;
          let netProfit = 0;
          const nowTime = new Date().toISOString().replace('T', ' ').substring(0, 19);

          while (this.data.positions.length > 0) {
            const pos = this.data.positions.pop();
            totalClosed++;
            netProfit += pos.profit;
            this.data.history.unshift({
              ticket: pos.ticket,
              time: nowTime,
              symbol: pos.symbol,
              type: pos.type,
              volume: pos.volume,
              open_price: pos.open_price,
              close_price: pos.current_price,
              profit: pos.profit
            });
          }

          this.data.account.balance += netProfit;
          this.updateAccountMetrics();
          this.renderPositions();
          this.renderHistory();
          this.renderAccountBar();

          const sign = netProfit >= 0 ? "+" : "";
          this.showToast(`Closed ${totalClosed} position(s). Realized P&L: ${sign}$${netProfit.toFixed(2)}`, "success");
        },

        async cancelOrder(ticket) {
          try {
            const res = await fetch(`${this.backendUrl}/trade/cancel`, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({ ticket: Number(ticket) }),
              signal: AbortSignal.timeout ? AbortSignal.timeout(3000) : undefined
            });
            if (res.ok) {
              this.showToast(`Order #${ticket} cancelled`, 'success');
              await this.refresh();
              return true;
            }
          } catch(e) {}

          const idx = this.data.orders.findIndex(o => o.ticket == ticket);
          if (idx !== -1) {
            this.data.orders.splice(idx, 1);
            this.renderOrders();
            this.showToast(`Pending order #${ticket} cancelled`, 'info');
            return true;
          }
          return false;
        },

        async cancelAllOrders() {
          if (this.data.orders.length === 0) {
            this.showToast("No pending orders to cancel", "info");
            return;
          }
          if (!confirm(`Cancel all ${this.data.orders.length} pending order(s)?`)) return;

          const count = this.data.orders.length;
          this.data.orders = [];
          this.renderOrders();
          this.showToast(`Cancelled all ${count} pending orders`, 'info');
        },

        addPosition(pos) {
          const normalized = this.normalizePosition(pos);
          this.data.positions.unshift(normalized);
          this.updateAccountMetrics();
          this.renderPositions();
          this.renderAccountBar();
          this.showToast(`New ${normalized.type} position #${normalized.ticket} opened for ${normalized.symbol}`, 'success');
        },

        addOrder(ord) {
          const normalized = this.normalizeOrder(ord);
          this.data.orders.unshift(normalized);
          this.renderOrders();
          this.showToast(`New pending ${normalized.type} order #${normalized.ticket} placed`, 'info');
        },

        showToast(message, type = 'info') {
          const container = document.getElementById("pm-toast-container");
          if (!container) return;

          const icons = { success: '✅', error: '❌', info: 'ℹ️' };
          const toast = document.createElement("div");
          toast.className = `pm-toast ${type}`;
          toast.innerHTML = `${icons[type] || 'ℹ️'}${message}`;
          container.appendChild(toast);

          setTimeout(() => {
            toast.style.opacity = '0';
            toast.style.transform = 'translateY(10px)';
            toast.style.transition = 'all 0.2s ease-out';
            setTimeout(() => toast.remove(), 250);
          }, 3200);
        }
      };

      window.addEventListener("DOMContentLoaded", () => {

          window.OrderPanel.init();

          window.PositionManager.init();

        // Null-guarded session table drag handler
        const dragEl = document.getElementById("session-table");
        if (dragEl) {
          let posX = 0, posY = 0, mouseX = 0, mouseY = 0;
          dragEl.onmousedown = e => {
            e.preventDefault();
            mouseX = e.clientX;
            mouseY = e.clientY;
            document.onmousemove = e2 => {
              posX = mouseX - e2.clientX;
              posY = mouseY - e2.clientY;
              mouseX = e2.clientX;
              mouseY = e2.clientY;
              dragEl.style.top = (dragEl.offsetTop - posY) + "px";
              dragEl.style.left = (dragEl.offsetLeft - posX) + "px";
            };
            document.onmouseup = () => {
              document.onmouseup = null;
              document.onmousemove = null;
            };
          };
        }
      });
